134void UartComm::parseLine(const QString &line)
135{
136 qDebug() << "[UART RAW] len=" << line.size() << " |" << line << "|";
137
138 if (line.isEmpty()) return;
139
140 if (line.startsWith('{')) {
141 parseJsonMeas(line);
142 return;
143 }
144
145 // ACK ignore silencieusement
146 if (line.startsWith("OK;"))
147 return;
148
149 // ERREUR
150 if (line.startsWith("ERR;")) {
151 int code = line.section(';', 1, 1).toInt();
152 emit errorOccurred("Erreur " + QString::number(code));
153 return;
154 }
155
156 // STATUS
157 if (line.startsWith("Parametre;")) {
158 QString params = line.mid(line.indexOf(';') + 1).replace(';', " ");
159 emit statusMessage("Parametre : " + params);
160 return;
161 }
162
163 if (line.startsWith("inductance=")) {
164 QString val = line.mid(line.indexOf('=') + 1);
165 emit statusMessage("Inductance : " + val);
166 return;
167 }
168
169 if (line.trimmed().toLower() == "simulation prete") {
170 emit statusMessage("Simulation prete");
171 return;
172 }
173
174 // MEAS
175 if (line.startsWith("MEAS;")) {
176 QStringList p = line.split(';');
177 if (p.size() == 5) {
178 Measurement m;
179 m.timestamp = QDateTime::currentDateTime();
180 m.voltageV = p[1].toDouble();
181 m.currentA = p[2].toDouble();
182 m.inductanceH = p[3].toDouble();
183 m.saturationCurrentA = p[4].toDouble();
184 emit newMeasurement(m);
185 }
186 return;
187 }
188
189 emit errorOccurred("Trame inconnue : " + line);
190}

200void UartComm::parseJsonMeas(const QString &json)
201{
202 QJsonParseError err;
203 QJsonDocument doc = QJsonDocument::fromJson(json.toUtf8(), &err);
204 if (err.error != QJsonParseError::NoError || !doc.isObject()) {
205 emit errorOccurred("JSON invalide : " + err.errorString());
206 return;
207 }
208
209 QJsonObject obj = doc.object();
210 QJsonArray intensites = obj.value("intensite").toArray();
211 QJsonArray temps = obj.value("temps").toArray();
212 QJsonArray tensions = obj.value("tension").toArray();
213
214 int n = qMin(intensites.size(), qMin(temps.size(), tensions.size()));
215
216 qDebug() << "[JSON] n=" << n;
217
218 for (int i = 0; i < n; i++)
219 {
220 Measurement m;
221 m.timestamp = QDateTime::currentDateTime();
222 m.currentA = intensites[i].toDouble();
223 m.voltageV = tensions[i].toDouble();
224 m.timeS = temps[i].toDouble();
225 m.saturationCurrentA = 0.0;
226
227 if (i < n - 1)
228 {
229 double dI = intensites[i + 1].toDouble() - intensites[i].toDouble();
230 double dt = temps[i + 1].toDouble() - temps[i].toDouble();
231 double dIdt = (dt > 0 ? dI / dt : 0);
232
233 if (qAbs(dIdt) > 1e-6)
234 m.inductanceH = qAbs(m.voltageV / dIdt);
235 else
236 m.inductanceH = 0.0;
237 }
238 else
239 {
240 double dI = intensites[i].toDouble() - intensites[i - 1].toDouble();
241 double dt = temps[i].toDouble() - temps[i - 1].toDouble();
242
243 if (dt > 0 && qAbs(dI) > 0.1)
244 m.inductanceH = qAbs(m.voltageV / (dI / dt));
245 else
246 m.inductanceH = 0.0;
247 }
248
249 emit newMeasurement(m);
250 }
