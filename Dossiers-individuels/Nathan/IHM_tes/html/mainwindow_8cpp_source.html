248void MainWindow::updateRealtimeDisplay(const Measurement &m)
249{
250 ui->valueI->setText(QString::number(m.currentA, 'f', 6) + " A");
251 ui->valueU->setText(QString::number(m.voltageV, 'f', 2) + " V");
252
253 if (m.inductanceH > 0)
254 m_lastValidL = m.inductanceH;
255 ui->valueL->setText(QString::number(m_lastValidL * 1e3, 'f', 3) + " mH");
256
257
258 double tGraph = m.timeS * 1000.0; // secondes -> ms
259 m_seriesCourant->append(tGraph, m.currentA);
260 m_seriesTension->append(tGraph, m.voltageV);
261 double L = (m.inductanceH > 1e-6) ? m.inductanceH : m_lastValidL;
262 m_seriesInductance->append(tGraph, L * 1e3);
263
264
265 // Axe X
266 static qint64 lastUpdate = 0;
267 qint64 now = QDateTime::currentMSecsSinceEpoch();
268
269 if (now - lastUpdate > 200) {
270 m_axisX->setRange(0, tGraph * 1.1);
271 lastUpdate = now;
272 }
273
274
275 // Calcul yMin/yMax sur toute la serie
276 double yMin = qMin(m.currentA, m.voltageV);
277 double yMax = qMax(m.currentA, m.voltageV);
278
279 for (const QPointF &p : m_seriesTension->points())
280 {
281 yMin = qMin(yMin, p.y());
282 yMax = qMax(yMax, p.y());
283 }
284 for (const QPointF &p : m_seriesCourant->points())
285 {
286 yMin = qMin(yMin, p.y());

291 yMin = qMin(yMin, p.y());
292 yMax = qMax(yMax, p.y());
293 }
294

296 m_axisY->setRange(yMin - margin, yMax + margin);

316void MainWindow::loadCsvIntoHistory()
317{
318 m_historyModel->removeRows(0, m_historyModel->rowCount());
319
320 QString path = QDir::homePath() + "/Documents/mesures.csv";
321 QFile file(path);
322
323 if (!file.open(QIODevice::ReadOnly | QIODevice::Text))
324 return;
325
326 QTextStream in(&file);
327
328 bool firstLine = true;
329 while (!in.atEnd())
330 {
331 QString line = in.readLine();
332 if (line.trimmed().isEmpty())
333 continue;
334
335 if (firstLine) {
336 firstLine = false;
337 continue;
338 }
339
340 QStringList parts = line.split(';');
341 if (parts.size() < 4)
342 continue;
343
344 QList<QStandardItem*> row;
345 for (const QString &p : parts)
346 row.append(new QStandardItem(p));
347
348 m_historyModel->appendRow(row);
349 }

359void MainWindow::onSearchTextChanged(const QString &text)
360{
361 for (int row = 0; row < m_historyModel->rowCount(); row++)
362 {
363 bool match = false;
364
365 for (int col = 0; col < m_historyModel->columnCount(); col++)
366 {
367 QString cell = m_historyModel->item(row, col)->text();
368 if (cell.contains(text, Qt::CaseInsensitive))
369 {
370 match = true;
371 break;
372 }
373 }
374
375 ui->tableHistory->setRowHidden(row, !match);
376 }
